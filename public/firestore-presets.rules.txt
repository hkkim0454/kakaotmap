rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Existing LED rule retained exactly: its expiry is NOT extended.
    match /{document=**} {
      allow read, write: if request.time < timestamp.date(2026, 8, 7);
    }

    // Only the two public Barogil slots can be read or changed.
    match /barogil_presets/{slot} {
      function validSlot() { return slot in ['1', '2']; }
      function validPlace() {
        let d = request.resource.data;
        return d.keys().hasAll(['name', 'address', 'x', 'y'])
          && d.keys().hasOnly(['name', 'address', 'x', 'y'])
          && d.name is string && d.name.size() > 0 && d.name.size() <= 120
          && d.address is string && d.address.size() <= 240
          && d.x is number && d.x >= -180 && d.x <= 180
          && d.y is number && d.y >= -90 && d.y <= 90;
      }
      allow get: if validSlot();
      allow create, update: if validSlot() && validPlace();
      allow delete: if validSlot();
    }
  }
}
